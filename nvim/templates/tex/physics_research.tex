\section{Core Field Equations}
The Dirac Lagrangian density for a free spin-$1/2$ field is given by:
\begin{equation}
\mathcal{L} = \bar{\psi}(i\gamma^\mu \partial_\mu - m)\psi
\end{equation}

Using bra-ket notation provided by the \texttt{physics} package:
\begin{equation}
\hat{H} \ket{\psi(t)} = i\hbar \pdv{}{t} \ket{\psi(t)}
\end{equation}

\section{Mathematical Derivations}

\begin{derivationbox}[Canonical Commutation Relations]
For scalar fields $\phi(x)$ and conjugate momenta $\pi(y)$:
\begin{align}
[\phi(\vec{x}, t), \pi(\vec{y}, t)] &= i\hbar \delta^{(3)}(\vec{x} - \vec{y}) \\
[\phi(\vec{x}, t), \phi(\vec{y}, t)] &= 0 \\
[\pi(\vec{x}, t), \pi(\vec{y}, t)] &= 0
\end{align}
\end{derivationbox}
